int CDlgStreamlines::TextureIteration(float* start, int alpha, float lapw)
{
	int n, x, y;
	int newx, newy, prevx, prevy;
	float vel_factor = m_VelDelta;

	for(x=0; x<256; x++)
		for(y=0; y<256; y++)
		{
			texture[256*y+x] = 0;

			prevx = x;
			prevy = y;

			// positive side
			for(n=1; n<=alpha; n++)
			{
				newx = prevx + FloatToInt(vel_factor*(float)mriData.u[256*prevy+prevx]);
				newy = prevy + FloatToInt(vel_factor*(float)mriData.v[256*prevy+prevx]);
				if(newx>255)newx=255;
				if(newy>255)newy=255;
				if(newx<0)newx=0;
				if(newy<0)newy=0;

				texture[256*y+x] = texture[256*y+x] + start[256*newy+newx];

				prevx = newx;
				prevy = newy;
				
			}

			prevx = x;
			prevy = y;

			// negative side
			for(n=1; n<=alpha; n++)
			{
				newx = prevx - FloatToInt(vel_factor*(float)mriData.u[256*prevy+prevx]);
				newy = prevy - FloatToInt(vel_factor*(float)mriData.v[256*prevy+prevx]);
				if(newx>255)newx=255;
				if(newy>255)newy=255;
				if(newx<0)newx=0;
				if(newy<0)newy=0;

				texture[256*y+x] = texture[256*y+x] + start[256*newy+newx];

				prevx = newx;
				prevy = newy;
				
			}

			// zero tau part
			texture[256*y+x] = texture[256*y+x] + start[256*y+x];
				
			// normalisation
			texture[256*y+x] = texture[256*y+x]/(2.0f*(float)alpha+1.0f);
		}

	// take laplacian
	float* text_lap = new float[256*256];
	TextureLaplacian(text_lap);

	for(x=0; x<256; x++)
		for(y=0; y<256; y++)
		{
			texture[256*y+x] = texture[256*y+x] + lapw*text_lap[256*y+x];
		}

	delete text_lap;

	return 0;
}

	Noise3D(wnoise_img, mriData3D.img_width, mriData3D.img_height, mriData3D.img_slices,
							  m_3DNoiseS+2, m_3DNoiseM*10.0f, m_3DNoiseV*10.0f, m_3DNoiseB);

	// add velocities to noise
	int x, y, z, index;
	short* mr = new short[mriData3D.img_width*mriData3D.img_height*mriData3D.img_slices];
	short* u = new short[mriData3D.img_width*mriData3D.img_height*mriData3D.img_slices];
	short* v = new short[mriData3D.img_width*mriData3D.img_height*mriData3D.img_slices];
	short* w = new short[mriData3D.img_width*mriData3D.img_height*mriData3D.img_slices];

	for(z=0; z<mriData3D.img_slices; z++)
		for(x=0; x<mriData3D.img_width; x++)
			for(y=0; y<mriData3D.img_height; y++)
			{
				index = z*mriData3D.img_width*mriData3D.img_height+y*mriData3D.img_width+x;
				
				// u part
				if(mriData3D.u[index] + unoise_img[index] > 1800)
					u[index] = FloatToInt(mriData3D.u[index] 
					+ unoise_img[index]) - 3600;
				else
					if(mriData3D.u[index] + unoise_img[index] < -1800)
						u[index] = FloatToInt(mriData3D.u[index] 
						+ unoise_img[index]) + 3600;
					else
						u[index] = FloatToInt(mriData3D.u[index] 
						+ unoise_img[index]);
			}

	for(z=0; z<mriData3D.img_slices; z++)
		for(x=0; x<mriData3D.img_width; x++)
			for(y=0; y<mriData3D.img_height; y++)
			{
				index = z*mriData3D.img_width*mriData3D.img_height+y*mriData3D.img_width+x;
				
				// v part
				if(mriData3D.v[index] + vnoise_img[index] > 1800)
					v[index] = FloatToInt(mriData3D.v[index] 
					+ vnoise_img[index]) - 3600;
				else
					if(mriData3D.v[index] + vnoise_img[index] < -1800)
						v[index] = FloatToInt(mriData3D.v[index] 
						+ vnoise_img[index]) + 3600;
					else
						v[index] = FloatToInt(mriData3D.v[index] 
						+ vnoise_img[index]);
			}
			
	for(z=0; z<mriData3D.img_slices; z++)
		for(x=0; x<mriData3D.img_width; x++)
			for(y=0; y<mriData3D.img_height; y++)
			{
				index = z*mriData3D.img_width*mriData3D.img_height+y*mriData3D.img_width+x;
				
				// w part
				if(mriData3D.w[index] + wnoise_img[index] > 1800)
					w[index] = FloatToInt(mriData3D.w[index] 
					+ wnoise_img[index]) - 3600;
				else
					if(mriData3D.w[index] + wnoise_img[index] < -1800)
						w[index] = FloatToInt(mriData3D.w[index] 
						+ wnoise_img[index]) + 3600;
					else
						w[index] = FloatToInt(mriData3D.w[index] 
						+ wnoise_img[index]);
					
							
			}
			
	mriData3DNoisy.SetData(mriData3D.img_width, mriData3D.img_height, mriData3D.img_slices, 
		mriData3D.mr, u, v, w);

	delete [] mr;
	delete [] u;
	delete [] v;
	delete [] w;
	delete [] unoise_img;
	delete [] vnoise_img;
	delete [] wnoise_img;

	MessageBeep(0);
	
}
